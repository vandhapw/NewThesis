% --- Font Setup ---
% Main font: Shinmyungjo (신명조) for Korean text.
% The following uses kotex's default HCR fonts as fallback if Shinmyungjo
% is not installed. For a true Shinmyungjo setup, install the font and use:
%   \SetHangulFonts{Shinmyungjo}{Shinmyungjo}{Shinmyungjo}
%   \SetHanjaFonts{Shinmyungjo}{Shinmyungjo}{Shinmyungjo}
% Fallback to Times Roman (ptm) for Latin characters.

\renewcommand{\sfdefault}{phv}

  {\endlist}

% --- Line Spacing ---
% 1.5 line spacing for readability (common for Korean theses)
\setstretch{1.5}

% --- Heading Styles ---
% Chapter title: 15 pt, bold, centered.
% [display] places label and title on separate lines (vertical stack).
\titleformat{\chapter}[display]
  {\normalfont\bfseries\centering\fontsize{15pt}{18pt}\selectfont}
  {\chaptername\ \thechapter}{3pt}{\fontsize{15pt}{18pt}\selectfont}
\titlespacing*{\chapter}{0pt}{0pt}{20pt}

% Section (subjudul): 13 pt, bold
\titleformat{\section}
  {\normalfont\bfseries\fontsize{13pt}{16pt}\selectfont}
  {\thesection}{1em}{}

% Subsection: 12 pt
\titleformat{\subsection}
  {\normalfont\fontsize{12pt}{14pt}\selectfont}
  {\thesubsection}{1em}{}

% Subsubsection: 11 pt (same as body)
\titleformat{\subsubsection}
  {\normalfont\fontsize{11pt}{13pt}\selectfont}
  {\thesubsubsection}{1em}{}

% --- Table and Figure Captions ---
% Table caption ABOVE the table, 10 pt, bold
\captionsetup[table]{
  position=above,
  font={small,bf},
  labelfont=bf,
  justification=raggedright
}

% Figure caption BELOW the figure, 10 pt, bold
\captionsetup[figure]{
  position=below,
  font={small,bf},
  labelfont=bf,
  justification=centering
}
